57 /*
58 *Internal function: SOH_FindMonitor implementation.
59 *@param[in] option for record type
60 *@param[out] true if control option record is in range, false otherwise
61 *@return pointer to monitor or NULL_PTR if request out of range or monitor not supported */
62static SOH_MonitorType* SOH_FindMonitor(VAR(Dcm_StartDataIn_RID_02C0_ClearSUMSOHLogs_DcmDspStartRoutineInSignal_0Type, AUTOMATIC) ControlOptionRecord, boolean* DidInRange)
63{
64 SOH_MonitorType* monitor = NULL_PTR;
65
66 switch(ControlOptionRecord)
67 {
68
69 case 0xF08F:
70 *DidInRange = TRUE;
71#if (SOH_IMMEDIATE_DELAYED_SUPPORT_ENABLED || SOH_DIAG_RESET_SUPPORT_ENABLED)
72 monitor = &SOH_ResetReliabilityMonitor;
73#endif
74 break;
75
76 case 0xF090:
77 *DidInRange = TRUE;
78
79#if (SOH_STATUS_NVM_SUPPORT_ENABLED || SOH_ACTION_NVM_SUPPORT_ENABLED || SOH_PRIVILEGE_NVM_SUPPORT_ENABLED|| \
80 SOH_EXCEP_OS_SUPPORT_ENABLED ||SOH_SERV_OS_SUPPORT_ENABLED || SOH_STACK_OS_SUPPORT_ENABLED || \
81 SOH_MEMPROT_OS_SUPPORT_ENABLED || SOH_TIMEPROT_OS_SUPPORT_ENABLED || SOH_ERR_OS_SUPPORT_ENABLED)
82 monitor = &SOH_MemoryStatusReliabilityMonitor;
83#endif
84 break;
85
86 case 0xF091:
87 *DidInRange = TRUE;
88# if(SOH_SHUTDOWN_DELAY_SUPPORT_ENABLED)
89 monitor = &SOH_PlatformReliabilityMonitor;
90#endif
91 break;
92
93 case 0xF092:
94 *DidInRange = TRUE;
95#if (SOH_CLEAR_DIAG_INFO_SUPPORT_ENABLED || SOH_DIAG_SESSION_SUPPORT_ENABLED || SOH_SEC_ACCESS_FAILURES_SUPPORT_ENABLED|| \
96 SOH_MAC_FAILURES_SUPPORT_ENABLED || SOH_SECP_ERRORS_SUPPORT_ENABLED )
97
98 monitor = &SOH_CommunicationReliabilityMonitor;
99#endif
100 break;
101
102 case 0xF093:
103 *DidInRange = TRUE;
104#if (SOH_LOCAL_SUP_FAILURES_SUPPORT_ENABLED || SOH_WdgMDEM_SUP_ERRORS_SUPPORT_ENABLED || SOH_ERRH_LOCAL_SUP_ERRORS_SUPPORT_ENABLED)
105 monitor = &SOH_WdgMReliabilityMonitor;
106#endif
107 break;
108
109 case 0xF094:
110 *DidInRange = TRUE;
111#if (ERRH_SOH_LOCAL_SUP_ERRORS_SUPPORT_ENABLED)
112 monitor = &SOH_WdgMReliabilityF094Monitor;
113#endif
114 break;
115
116 case 0xFFFF:
117 *DidInRange = TRUE;
118#if (NUM_OF_DETECTORS != 0u)
119 monitor = &SOH_AllMonitors;
120#endif
121 break;
122
123 default:
124 *DidInRange = FALSE;
125 break;
126 }
127
128 return monitor;
129}

191/*
192 * This function is called only by Runnable_SOH_RIDClearLogs() as part of the RID 0x02C0 Clear_SUM_SOH_logs.
193 *
194 * \Requirement{033}
195 * It calls SOH_ClearItems() to clear the detection counters, the time windows and the DID counters in the SOH_MonitorType instances.
196 * \Satisfies{019}
197 */
198static FUNC(Std_ReturnType, SUM_SOH_CODE) ComputeReturnStatus(VAR(Dcm_OpStatusType, AUTOMATIC) OpStatus,
199 SOH_MonitorType const* monitor)
200{
201
202 Std_ReturnType ReturnStatus=E_NOT_OK;
203 Std_ReturnType NvMStatus;
204
205 switch(OpStatus)
206 {
207 /*DCM Initial phase, if no write pending, clean the monitor, return PENDING*/
208 /*DCM Pending phase, check that the writes have been performed.
209 * If there is any write pending previously, return PENDING
210 * or Start clearing items and return PENDING
211 * Or check NVM status and return OK or NOT_OK
212 * */
213 case DCM_INITIAL :
214 /* Check if at least one NvM write status is pending */
215 NvMStatus = SOH_CheckNvMWritesList(monitor->SOH_Detectors, monitor->length);
216 if(NvMStatus != SOH_E_PENDING)
217 {
218 /*[$Satisfies $SDD_SOH 015]*/
219
220 /* Notify that this runnable is updating the NvM */
221 Rte_IrvWrite_Runnable_SOH_RIDClearLogs_SOH_SOHRIDIsWriting((TRUE));
222
223 /* [$Satisfies $DDS_SOH 033] */
224 /* No NvM activity pending, so clear all the metrics and write them in the NvM */
225 NvMStatus = SOH_ClearItems(monitor->SOH_Detectors, monitor->length);
226
227 /*If the writes have started, return PENDING */
228 if(NvMStatus==E_OK)
229 {
230 ReturnStatus = SOH_E_PENDING;
231 }
232 }
233 else
234 {
235 ReturnStatus = E_NOT_OK;
236 }
237 break;
238 case DCM_PENDING :
239 /*Check if the NvM writing has been completed*/
240 ReturnStatus = SOH_CheckNvMWritesList(monitor->SOH_Detectors, monitor->length);
241 break;
242 /* If the state is invalid, return NOT_OK*/
243 default:
244 ReturnStatus=E_NOT_OK;
245 break;
246 }
247
248 /*Notify that this runnable is not updating the NvM ( No NvM write is ongoing)*/
249 if(ReturnStatus!=SOH_E_PENDING)
250 {
251 Rte_IrvWrite_Runnable_SOH_RIDClearLogs_SOH_SOHRIDIsWriting((FALSE));
252 }
253
254 return ReturnStatus;
255
256}

258 /*
259 *Internal function: SOH_ConvertUint16ToBigEndianUint8Array implementation.
260 *
261 * It converts an uint16 to an uint8 two byte array big endian ordered
262 *
263 *@param[in] source
264 *@param[out] output
265 *@return void
266 */
267static void SOH_ConvertUint16ToBigEndianUint8Array(uint16 source, CONSTP2VAR(uint8, AUTOMATIC, RTE_APPL_DATA) output )
268{
269 /* to Big-Endian */
270 *output = (uint8) ((source & 0xFF00u) >> 8u);
271 *(output+1) = (uint8) ((source & 0x00FFu) );
272}
