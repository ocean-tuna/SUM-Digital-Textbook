110 FUNC(Std_ReturnType, SUM_SOH_CODE) SOH_CheckNvMWritesList(SOH_DetectorType* const items[], uint8 length)
111{
112 Std_ReturnType NvmStatus;
113 uint8 i;
114 boolean AllOkFlag = TRUE;
115
116 for(i = 0u; i < length; i++)
117 {
118 /*Check the Nvm write status for each detector*/
119
120 NvmStatus = SOH_CheckNvMWrite(items[i]);
121 if(NvmStatus != E_OK)
122 {
123 /* Not all NvM calls returned NVM_REQ_OK*/
124 AllOkFlag =FALSE;
125 }
126 if(NvmStatus == SOH_E_PENDING)
127 {
128 /*At least one NvM writing is pending*/
129 break;
130 }
131
132 }
133
134 if(AllOkFlag == TRUE)
135 {
136 /*If all the NvM calls were OK, return E_OK*/
137 NvmStatus = E_OK;
138 }
139 else if(NvmStatus != SOH_E_PENDING)
140 {
141 /*If at least one NvM call was not ok and there are not
142 * pending NvM writing return SOH_E_PENDING
143 */
144 NvmStatus = E_NOT_OK;
145 }
146 else
147 {
148 /*Do Nothing*/
149 }
150
151 return NvmStatus;

160 FUNC(Std_ReturnType, SUM_SOH_CODE) SOH_CheckNvMWrite(SOH_DetectorType const* item)
161 {
162
163 Std_ReturnType ReturnStatus;
164 NvM_RequestResultType NvmErrorStatus;
165 Std_ReturnType NvmStatus = E_NOT_OK;
166#if (NUM_OF_SOH_DETECTORS != 0u)
167 if(item->NvMDID.SOH_RteCallNvmGetErrorStatus_fp != NULL_PTR)
168 {
169
170 ReturnStatus = item->NvMDID.SOH_RteCallNvmGetErrorStatus_fp(&NvmErrorStatus);
171 /*If the return Status is E_OK and the NvmErrorStatus is OK:
172 * -if the SOH_Storage is not AdaptiveNvM or SecureCumulativeNvM return E_OK
173 * -otherwise, check if there is any NvM writing pending related to the Ctr
174 * and:
175 * -if the return status is OK and NvMErrorStatus is OK, return E_OK
176 * -if the NvM writing is pending return SOH_E_PENDING
177 * -otherwise return E_NOT_OK
178 *
179 */
180 if ((ReturnStatus == E_OK) && (NvmErrorStatus == NVM_REQ_OK))
181 {
182 if((item->SOH_Storage == SOH_AdaptiveNvM) || (item->SOH_Storage == SOH_SecureCumulativeNvM))
183 {
184 ReturnStatus = item->NvMCtr.SOH_RteCallNvmGetErrorStatus_fp(&NvmErrorStatus);
185
186 if((ReturnStatus == E_OK) && (NvmErrorStatus == NVM_REQ_OK))
187 {
188 NvmStatus = E_OK;
189 }
190 else if ((ReturnStatus == E_OK) && (NvmErrorStatus == NVM_REQ_PENDING))
191 {
192 NvmStatus = SOH_E_PENDING;
193 }
194 else
195 {
196 /*Do Nothing*/
197 }
198 }
199 else
200 {
201 NvmStatus = E_OK;
202 }
203 }
204 else if((ReturnStatus == E_OK) && (NvmErrorStatus == NVM_REQ_PENDING))
205 {
206 /*If the NvM writing is pending, return SOH_E_PENDING*/
207 NvmStatus = SOH_E_PENDING;
208 }
209 else if((item->SOH_Storage == SOH_AdaptiveNvM) || (item->SOH_Storage == SOH_SecureCumulativeNvM))
210 {
211 /*If the SOH_Storage is AdaptiveNvM or SecureCumulativeNvM, check if there
212 * is any NvM writing pending related to the Cntr
213 */
214 ReturnStatus = item->NvMCtr.SOH_RteCallNvmGetErrorStatus_fp(&NvmErrorStatus);
215
216 /*If yes, return SOH_E_PENDING, otherwise return E_NOT_OK*/
217 if((ReturnStatus == E_OK) && (NvmErrorStatus == NVM_REQ_PENDING))
218 {
219 NvmStatus = SOH_E_PENDING;
220 }
221 }
222 else
223 {
224 /*Do nothing*/
225 }
226 }
227 else
228 {
229 #if (ERRH_SOH_LOCAL_SUP_ERRORS_SUPPORT_ENABLED == 0x01)
230 /* Case DID 0xF94. No way to be informed from the ERRH about the status of its Nvm ram block.
231 Assuming that the last nvm write was successful */
232 if (item->SOH_Filter == SOH_ERRHImplementsFilter)
233 {
234 NvmStatus = E_OK;
235 }
236 #else
237 /* Do nothing. NvmStatus = E_NOT_OK*/
238 #endif
239
240 }
241
242#endif
243 return NvmStatus;
244 }

281 FUNC(Std_ReturnType, SUM_SOH_CODE) SOH_Write(SOH_DetectorType const* item)
282{
283 Std_ReturnType retValue;
284
285 if(item->NvMDID.NvmRamBlock==NULL_PTR)
286 {
287 /*If the detector is the one related to DID F094
288 * ask the ERRH to write the NvM to zero
289 * */
290#if ERRH_SOH_LOCAL_SUP_ERRORS_SUPPORT_ENABLED
291 /* Clear operation (by setting the the detectors to their default values) done in the SSFM module*/
292 retValue = Rte_Call_SUM_SOH_RP_SOH_ERRH_ClearRID0x02C0_Start_Clear();
293#endif
294 }
295 else
296 {
297 *(item->NvMDID.NvmRamBlock) = *(item->SOH_TresholdItem.SOH_DIDCntr);
298 item->NvMDID.SOH_RteCallNvmSetRamBlockStatus_fp(TRUE);
299 retValue = item->NvMDID.SOH_RteCallNvmWriteBlock_fp(NULL_PTR);
300 if((retValue == E_OK) &&((item->SOH_Storage == SOH_AdaptiveNvM) || (item->SOH_Storage == SOH_SecureCumulativeNvM)))
301 {
302 *(item->NvMCtr.NvmRamBlock) = *(item->SOH_TresholdItem.SOH_Cntr);
303 item->NvMCtr.SOH_RteCallNvmSetRamBlockStatus_fp(TRUE);
304 retValue = item->NvMCtr.SOH_RteCallNvmWriteBlock_fp(NULL_PTR);
305 }
306 }
307
308 return retValue;
309}

362 FUNC(void, SUM_SOH_CODE) SOH_UpdateTimeWindows(uint32 counterValue)
363{
364 uint8 i;
365 uint32 *previousCounterValue;
366
367 for(i=0u;i<NUM_OF_DETECTORS;i++)
368 {
369 /* [$Satisfies $DDS_SOH 018] */
370 if((SOH_DetectorItems[i]->SOH_Filter == SOH_TWFiltering) &&
371 ((*(SOH_DetectorItems[i]->SOH_TresholdItem.SOH_Cntr)) != 0x00u))
372 {
373 previousCounterValue = SOH_DetectorItems[i]->SOH_TresholdItem.SOH_PrevCounterValue;
374
375 /* At init and for detectors with data in NvM */
376 if( (*previousCounterValue==0x00u) || (*previousCounterValue>counterValue) )
377 {
378 /* [$Satisfies $DDS_SOH 034] */
379 *previousCounterValue = counterValue;
380 }
381
382 /* Update Time Windows if 1s elapsed */
383 /* PRQA S 3469 3 *//* It's an OS limit */
384 while( OS_TICKS2MS_SOH_Counter(counterValue-*previousCounterValue) >= 1000u )
385 {
386 /* Increase *previousCounterValue by the number of ticks in 1s */
387 *previousCounterValue = *previousCounterValue + (1000000000u/OSTICKDURATION_SOH_Counter);
388
389 /* Increment the time window of 1s */
390 /* [$Satisfies $DDS_SOH 019] */
391 (*(SOH_DetectorItems[i]->SOH_TresholdItem.SOH_TW))++;
392
393 /*If the detection counter does not reach its threshold count calibration within the
394 * calibration time window, the SUM_SOH resets its time window to 0x000000 to start
395 * a new detection cycle and resets its detection counter to 0x0000*/
396 if((*(SOH_DetectorItems[i]->SOH_TresholdItem.SOH_TW)) >= SOH_DetectorItems[i]->SOH_TresholdItem.SOH_TWThreshold())
397 {
398 /* [$Satisfies $DDS_SOH 030] */
399 *(SOH_DetectorItems[i]->SOH_TresholdItem.SOH_TW) = 0x00000000;
400 /* [$Satisfies $DDS_SOH 029] */
401 *(SOH_DetectorItems[i]->SOH_TresholdItem.SOH_Cntr) = 0x00u;
402 *previousCounterValue = 0x00u;
403
404 break;

480 FUNC(boolean, SUM_SOH_CODE) SOH_Filter(SOH_DetectorType const* item)
481{
482 boolean ret = TRUE;
483
484 /*In the case where the detection counter reaches its threshold count calibration, the following steps SUM_SOH:
485 - Increments the DID counter by one if it has not reached the MAX value.
486 - Resets the detection counter to 0x0000
487 */
488 if((*(item->SOH_TresholdItem.SOH_Cntr))>= item->SOH_TresholdItem.SOH_CntrThreshold())
489 {
490 /* [$Satisfies $DDS_SOH 021] */
491 *(item->SOH_TresholdItem.SOH_Cntr) = 0x0000;
492
493 if(item->SOH_Filter == SOH_TWFiltering)
494 {
495 /* [$Satisfies $DDS_SOH 022] */
496 *(item->SOH_TresholdItem.SOH_TW) = 0x00000000;
497 *(item->SOH_TresholdItem.SOH_PrevCounterValue) = 0x00u;
498 }
499
500 if(*(item->SOH_TresholdItem.SOH_DIDCntr)<0xFFFEu)
501 {
502 /* [$Satisfies $DDS_SOH 020] */
503 (*(item->SOH_TresholdItem.SOH_DIDCntr))++;
504 }
505 }
506 else
507 {
508 ret = FALSE;
509 }
510
511 return ret;
512}

520 FUNC(void, SUM_SOH_CODE) SOH_CheckConditionAndWrite(SOH_DetectorType const* item)
521 {
522 boolean DIDCnt_Condition_b = FALSE;
523 boolean DetectionCnt_Condition_b = FALSE;
524
525 DisableWrite DisableWriteData;
526 Std_ReturnType NvmStatus;
527
528 static uint16 prev_DIDCnt_u16 = 0;
529 static uint16 prev_DetectionCnt_u16 = 0;
530 uint16 curr_DIDCnt_u16;
531 uint16 curr_DetectionCnt_u16;
532
533 /* Get previous DID Counter value */
534 prev_DIDCnt_u16 = *(item->NvMDID.NvmRamBlock);
535
536 /* Get current DID Counter value after processed by corresponding filter */
537 curr_DIDCnt_u16 = *(item->SOH_TresholdItem.SOH_DIDCntr);
538
539 /* Check whether DID counter is changed */
540 if((prev_DIDCnt_u16 != curr_DIDCnt_u16))
541 {
542 DIDCnt_Condition_b = TRUE;
543 }
544
545 /*[$Satisfies $SDD_SOH 058]*/
546 if((item->SOH_Storage == SOH_AdaptiveNvM) || (item->SOH_Storage == SOH_SecureCumulativeNvM))
547 {
548 /* Get previous Detection Counter value */
549 prev_DetectionCnt_u16 = *(item->NvMCtr.NvmRamBlock);
550
551 /* Get current Detection Counter value after processed by corresponding filter */
552 curr_DetectionCnt_u16 = *(item->SOH_TresholdItem.SOH_Cntr);
553
554 /* Check whether Detection counter is changed */
555 if(prev_DetectionCnt_u16 != curr_DetectionCnt_u16)
556 {
557 DetectionCnt_Condition_b = TRUE;
558 }
559 }
560
561 /* If at least one filed must be updated */
562 if((TRUE == DIDCnt_Condition_b) || (TRUE == DetectionCnt_Condition_b))
563 {
564 /* get NvM Writes status if any write process is pending*/
565 NvmStatus = SOH_CheckNvMWrite(item);
566
567 /* If no NvM writes pending*/
568 if(SOH_E_PENDING != NvmStatus)
569 {
570 /* Read Voltage operation range from SUM_SUSD to determine whether Voltage in range*/
571 (void)Rte_Read_SUSD_SOH_BatteryVoltage_DisableNVMWrite_DisableWriteData(&DisableWriteData);
572
573 /* If Battery Voltage Operating in range */
574 if(DisableWriteData == FALSE)
575 {
576 /* Perform write counters into memory */
577 (void)SOH_Write(item);
578
579 }
580 }
581 }
